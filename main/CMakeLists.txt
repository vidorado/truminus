# Embed C6 slave firmware if it has been built (run 'make build-c6' first).
# When present the binary is streamed to the C6 co-processor at startup via
# SDIO OTA if the version does not match the host ESP-Hosted library.
set(C6_FW_BIN "${CMAKE_CURRENT_SOURCE_DIR}/slave_fw/network_adapter.bin")
if(EXISTS "${C6_FW_BIN}")
    set(C6_FW_EMBED "slave_fw/network_adapter.bin")
    message(STATUS "TruMinus: C6 slave firmware found — will be embedded")
else()
    set(C6_FW_EMBED "")
    message(STATUS "TruMinus: C6 slave firmware absent — run 'make build-c6' then rebuild")
endif()

# Build a LittleFS image from <project>/data/ and flash it to the `littlefs`
# partition defined in partitions_16MB.csv.  FLASH_IN_PROJECT hooks the image
# into `idf.py flash` automatically; to flash *only* the filesystem image use
# `idf.py littlefs-flash-littlefs` (target name generated by the helper).
littlefs_create_partition_image(
    littlefs
    "${CMAKE_SOURCE_DIR}/data"
    FLASH_IN_PROJECT
)

target_compile_options(${COMPONENT_LIB} PRIVATE
    "-include" "${CMAKE_CURRENT_SOURCE_DIR}/core/flags.h"
)

if(EXISTS "${C6_FW_BIN}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE C6_FW_AVAILABLE=1)
endif()

target_compile_definitions(${COMPONENT_LIB} PRIVATE ENABLE_BLE=1)

# Route the IDF panic handler through crashcatch.cpp's __wrap_ shim so it can
# stash the crash context into RTC memory before the normal panic/reboot runs.
# panic_handler.c calls esp_panic_handler() by name, so --wrap intercepts it.
target_link_libraries(${COMPONENT_LIB} INTERFACE "-Wl,--wrap=esp_panic_handler")
